```python
#!/usr/bin/env python3
"""Stage generated task inputs: instruction.md and the fixture build context.

Regenerates from source-of-truth (evals.json + run_arm.py constants) so the
Harbor task can never drift from the seatbelt harness's framing.
"""
import importlib.util
import json
import shutil
import stat
import subprocess
from pathlib import Path

SPIKE = Path(__file__).resolve().parent
WORKSPACE = SPIKE.parent
FIXTURE = WORKSPACE / "fixtures" / "ticketd"
TASK = SPIKE / "eval-0"
CONTEXT = TASK / "environment" / "context"

def load_run_arm():
    spec = importlib.util.spec_from_file_location(
        "run_arm", WORKSPACE / "evals" / "isolation" / "run_arm.py")
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod

def main():
    run_arm = load_run_arm()
    evals = json.loads((WORKSPACE / "evals" / "evals.json").read_text())
    ev = next(e for e in evals["evals"] if e["id"] == 0)
    (TASK / "instruction.md").write_text(ev["prompt"] + run_arm.AUTONOMY_FRAMING + "\n")

    if CONTEXT.exists():
        for p in CONTEXT.rglob("*"):          # fixture legacy trees can be read-only
            p.chmod(p.stat().st_mode | stat.S_IWUSR)
        shutil.rmtree(CONTEXT)
    CONTEXT.mkdir(parents=True)
    shutil.copytree(FIXTURE, CONTEXT / "ticketd", symlinks=True)
    head = subprocess.run(["git", "-C", str(CONTEXT / "ticketd"), "rev-parse", "HEAD"],
                          capture_output=True, text=True, check=True).stdout.strip()
    print(f"staged ticketd at HEAD {head}")

if __name__ == "__main__":
    main()
```

```python
#!/usr/bin/env python3
"""Wrap the unmodified grade_checks.py into Harbor's reward contract.

Reads grade_checks JSON {check: {"ok": true|false|null, "evidence": str}} and
writes reward.json as a flat float dict. null (not mechanically scoreable)
counts 0.0 and is tallied in `unscoreable` so the report can flag it.
"""
import argparse
import json
import subprocess
import sys
from pathlib import Path

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--outputs", required=True)
    ap.add_argument("--reward-dir", default="/logs/verifier")
    ap.add_argument("--grader", default="/grader/grade_checks.py")
    args = ap.parse_args()

    proc = subprocess.run(
        [sys.executable, args.grader, "--eval", "0", "--outputs", args.outputs],
        capture_output=True, text=True)
    print(proc.stdout)
    print(proc.stderr, file=sys.stderr)
    checks = json.loads(proc.stdout)

    reward = {name: (1.0 if res.get("ok") is True else 0.0)
              for name, res in checks.items()}
    reward["total"] = float(sum(1 for r in checks.values() if r.get("ok") is True))
    reward["unscoreable"] = float(sum(1 for r in checks.values() if r.get("ok") is None))

    out = Path(args.reward_dir) / "reward.json"
    out.parent.mkdir(parents=True, exist_ok=True)
    out.write_text(json.dumps(reward, indent=1))
    print(f"wrote {out}: total={reward['total']} unscoreable={reward['unscoreable']}")

if __name__ == "__main__":
    main()
```

```python
    """Copy an iteration-2 arm's outputs and re-attach git history from bundles."""
    import shutil, stat
    src = IT2 / arm / "outputs"
    dst = tmp_path / arm
    shutil.copytree(src, dst, symlinks=True)
    for p in dst.rglob("*"):
        p.chmod(p.stat().st_mode | stat.S_IWUSR)
    manifest = json.loads((IT2.parent / "_git_bundles" / "manifest.json").read_text())
    git_bundles = SPIKE.parent / "evals" / "git_bundles.py"

        repo_rel = key.split(f"{arm}/outputs/", 1)[-1]
        target = dst / repo_rel
        if bundle and target.exists():
            subprocess.run([sys.executable, str(git_bundles), "graft",
                            str(bundle), str(target)], check=True)
```

```python
RkBaseline is the stock ClaudeCode adapter plus the guard hook (mode=block).
RkWithSkill additionally uploads the skill into the container at a neutral
path and appends the same prompt suffix run_arm.py uses — the container-era
equivalent of the seatbelt profile's skill-subtree re-allow.

from __future__ import annotations

import json
import tempfile
from pathlib import Path
from typing import override

from harbor.agents.installed.claude_code import ClaudeCode

SPIKE = Path(__file__).resolve().parents[1]
WORKSPACE = SPIKE.parent
REPO = WORKSPACE.parent
SKILL_SOURCE_DIR = REPO / "rebuild-kit"
GUARD_HOOK_SOURCE = WORKSPACE / "evals" / "isolation" / "guard_hook.py"

SKILL_CONTAINER_DIR = "/opt/wsvc-kit"

WITH_SKILL_SUFFIX = (

    def _settings_payload(self) -> dict:
```

```python
#!/usr/bin/env python3
"""Pre-flight isolation gate for the Harbor spike (ports verify_isolation.py's
spirit to containers). Fails loudly; a paid run must not start unless this
exits 0. Checks are static; the live guard-hook check happens in the shakedown
trial (Task 5) because it needs a real agent process.
"""
import argparse
import subprocess
import sys
import tomllib
from pathlib import Path

SPIKE = Path(__file__).resolve().parent

# Names/strings that would mean skill or assertion content leaked into the
# agent image. render_guide.py & scaffold.py are rebuild-kit script names.
MARKERS = ["SKILL.md", "evals.json", "render_guide.py", "scaffold.py",
           "rebuild-kit", "grade_checks"]

def sh(cmd: list[str]) -> tuple[int, str]:
    r = subprocess.run(cmd, capture_output=True, text=True)
    return r.returncode, r.stdout + r.stderr

def fail(msg: str):
    print(f"ISOLATION FAIL: {msg}")
    sys.exit(1)

def probe_image(image: str, markers: list[str]):
    name_expr = " -o ".join(f'-name "{m}"' for m in markers if "/" not in m)
    code, out = sh(["docker", "run", "--rm", image, "sh", "-c",
                    f"find / \\( {name_expr} \\) -not -path '/proc/*' 2>/dev/null; true"])
    if code != 0:
        fail(f"cannot probe image {image}: {out}")
    hits = [l for l in out.splitlines() if l.strip()]
    if hits:
        fail(f"marker files present in {image}: {hits[:5]}")
    grep_targets = "/opt /app /root /home /installed-agent /tests /grader"
    code, out = sh(["docker", "run", "--rm", image, "sh", "-c",
                    f"grep -rl 'rebuild-kit' {grep_targets} 2>/dev/null; true"])
    hits = [l for l in out.splitlines() if l.strip()]
    if hits:
        fail(f"'rebuild-kit' string present in {image}: {hits[:5]}")

def check_task_toml():
    cfg = tomllib.loads((SPIKE / "eval-0" / "task.toml").read_text())
    if cfg.get("verifier", {}).get("environment_mode") != "separate":
        fail("verifier is not in separate mode — assertions would enter the agent container")
    env = cfg.get("environment", {})
    for key in ("mounts", "volumes"):
        if env.get(key):
            fail(f"task.toml declares host {key} — forbidden")

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--agent-image", default="wsvc-ev0:local")
    ap.add_argument("--verifier-image", default="wsvc-ev0-verifier:local")
    args = ap.parse_args()

    check_task_toml()
    probe_image(args.agent_image, MARKERS)
    # The verifier image legitimately holds the grader; it must still never
    # hold the skill.
    probe_image(args.verifier_image, ["SKILL.md", "render_guide.py", "scaffold.py"])
    print("isolation checks passed")

if __name__ == "__main__":
    main()
```

```python
"""Fold Harbor trial outputs into the iteration-style parity report."""
from __future__ import annotations

import argparse
import json
import statistics
from dataclasses import dataclass, field
from pathlib import Path

SPIKE = Path(__file__).resolve().parents[1]
IT2_EVAL0 = SPIKE.parent / "iteration-2-sonnet" / "eval-0-full-generation"

@dataclass
class ArmResults:
    name: str
    scored: list[dict] = field(default_factory=list)       # reward.json dicts
    infra_failures: list[str] = field(default_factory=list)

    def mean_total(self) -> float:
        return statistics.mean(r["total"] for r in self.scored) if self.scored else 0.0

    def check_pass_rate(self) -> dict[str, float]:
        keys = [k for k in (self.scored[0] if self.scored else {}) if k not in ("total", "unscoreable")]
        return {k: statistics.mean(r.get(k, 0.0) for r in self.scored) for k in keys}

@dataclass
class Verdict:
    with_skill_mean: float
    baseline_mean: float
    baseline_reference: float
    gap: float
    passed: bool

def collect_trials(job_dir: Path) -> dict[str, ArmResults]:
    arms: dict[str, ArmResults] = {}
    for trial in sorted(p for p in job_dir.iterdir() if (p / "config.json").exists()):
        cfg = json.loads((trial / "config.json").read_text())
        agent = (cfg.get("agent") or {})
        name = agent.get("name") or str(agent.get("import_path", "unknown")).split(":")[-1]
        arm = arms.setdefault(name, ArmResults(name))
        result = json.loads((trial / "result.json").read_text()) if (trial / "result.json").exists() else {}
        reward_path = trial / "verifier" / "reward.json"
        if result.get("exception_info") or not reward_path.exists():
            arm.infra_failures.append(trial.name)
        else:
            arm.scored.append(json.loads(reward_path.read_text()))
    return arms

def parity_verdict(arms: dict[str, ArmResults], baseline_reference_total: float) -> Verdict:
    ws = arms.get("rk-with-skill", ArmResults("rk-with-skill"))
    bl = arms.get("rk-baseline", ArmResults("rk-baseline"))
    gap = ws.mean_total() - bl.mean_total()
    passed = (ws.mean_total() >= 9.0
              and abs(bl.mean_total() - baseline_reference_total) <= 1.5
              and gap >= 4.0)
    return Verdict(ws.mean_total(), bl.mean_total(), baseline_reference_total, gap, passed)

def iteration2_reference() -> float:
    stored = json.loads((IT2_EVAL0 / "without_skill" / "mechanical_checks.json").read_text())
    return float(sum(1 for r in stored.values() if r.get("ok") is True))

def render_markdown(arms: dict[str, ArmResults], verdict: Verdict) -> str:
    lines = ["# Harbor spike — eval-0 parity report", ""]
    lines.append(f"**Parity verdict: {'PASS' if verdict.passed else 'FAIL'}** — "
                 f"with-skill mean {verdict.with_skill_mean:.1f}, baseline mean "
                 f"{verdict.baseline_mean:.1f} (iteration-2 mechanical reference "
                 f"{verdict.baseline_reference:.1f}), gap {verdict.gap:.1f}.")
    for arm in arms.values():
        lines += ["", f"## {arm.name}",
                  f"- scored trials: {len(arm.scored)}; infra failures (unscored, rerun): "
                  f"{len(arm.infra_failures)} {arm.infra_failures}",
                  f"- mean total: {arm.mean_total():.2f}", "", "| check | pass rate |", "|---|---|"]
        lines += [f"| {k} | {v:.0%} |" for k, v in sorted(arm.check_pass_rate().items())]
    return "\n".join(lines) + "\n"

def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--job-dir", required=True)
    ap.add_argument("--out", default="-")
    args = ap.parse_args()
    arms = collect_trials(Path(args.job_dir))
    verdict = parity_verdict(arms, baseline_reference_total=iteration2_reference())
    md = render_markdown(arms, verdict)
    if args.out == "-":
        print(md)
    else:
        Path(args.out).write_text(md)

if __name__ == "__main__":
    main()
```
